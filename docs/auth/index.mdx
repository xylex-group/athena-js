## Base setup

```ts
import { createClient } from "@xylex-group/athena"

// Proxy mode (same-origin /api/auth):
export const client = createClient({
  key: "gateway_api_key",
  auth: { url: "http://localhost:3001/api/auth" },
})

// Direct upstream: use @xylex-group/athena-auth-ui createAthenaAuthClient
// with appendAuthPath: false — see auth-routing-proxy-and-direct-upstream.md
```
